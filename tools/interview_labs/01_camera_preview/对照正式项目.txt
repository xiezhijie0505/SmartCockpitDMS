    // 【对应实验步骤2】参数：和 lab 几乎一样
    m_cap.set(CAP_PROP_FOURCC, MJPG);
    m_cap.set(CAP_PROP_FRAME_WIDTH, 320);

    m_cap.set(CAP_PROP_BUFFERSIZE, 1);

    // 【难点】两个定时器：

    //   m_recognizeTimer ~200ms → 识别/活体  「能认」
    m_timer->start(33);
    m_recognizeTimer->start(200);
}
